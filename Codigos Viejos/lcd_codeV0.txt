#include <UTFTGLUE.h>
#include "imagenes.h"

UTFTGLUE myGLCD(0, 13, 12, 33, 32, 15); // Constructor compatible con tus pines [cite: 56]

// Factor de escala: cada caracter se convertirá en un bloque sólido de 6x6 píxeles
const int ESCALA_PIXEL = 6; 

void setup() {
  myGLCD.InitLCD();      // Inicializa pantalla [cite: 59]
  myGLCD.setRotation(3); // Forzar modo horizontal (480x320) [cite: 54]
}

void loop() {
  // --- MOSTRAR EL SOL ---
  myGLCD.fillScr(255, 255, 255); // Fondo blanco limpio
  // Centrado: X = (480 - (40 * 6)) / 2 = 120 | Y = (320 - (40 * 6)) / 2 = 40
  dibujarIconoHD(sol_map, 120, 40, 255, 204, 0); // Amarillo brillante
  delay(3000);

  // --- MOSTRAR EL OSO ---
  myGLCD.fillScr(255, 255, 255);
  dibujarIconoHD(oso_map, 120, 40, 140, 80, 35); // Marrón pelaje
  delay(3000);

  // --- MOSTRAR LA SAL ---
  myGLCD.fillScr(255, 255, 255);
  dibujarIconoHD(sal_map, 120, 40, 180, 185, 190); // Gris texturizado para el salero
  delay(3000);

  // --- MOSTRAR EL ALA ---
  myGLCD.fillScr(255, 255, 255);
  dibujarIconoHD(ala_map, 120, 40, 100, 200, 240); // Celeste cielo para el ala
  delay(3000);

  // --- MOSTRAR EL OJO ---
  myGLCD.fillScr(255, 255, 255);
  dibujarIconoHD(ojo_map, 120, 40, 30, 110, 220); // Azul profundo para el iris
  delay(3000);
}

// Función que lee los bloques y los proyecta con fillRect nativo de tu demo
void dibujarIconoHD(const char* mapa[], int x_inicio, int y_inicio, int r, int g, int b) {
  for (int y = 0; y < FILAS; y++) {
    for (int x = 0; x < COLUMNAS; x++) {
      char caracter = mapa[y][x];

      if (caracter == 'X') {
        myGLCD.setColor(0, 0, 0); // Contornos negros definidos [cite: 69]
        myGLCD.fillRect(x_inicio + (x * ESCALA_PIXEL), 
                        y_inicio + (y * ESCALA_PIXEL), 
                        x_inicio + (x * ESCALA_PIXEL) + ESCALA_PIXEL - 1, 
                        y_inicio + (y * ESCALA_PIXEL) + ESCALA_PIXEL - 1);
      }
      else if (caracter == '#') {
        myGLCD.setColor(r, g, b); // Color dinámico de relleno
        myGLCD.fillRect(x_inicio + (x * ESCALA_PIXEL), 
                        y_inicio + (y * ESCALA_PIXEL), 
                        x_inicio + (x * ESCALA_PIXEL) + ESCALA_PIXEL - 1, 
                        y_inicio + (y * ESCALA_PIXEL) + ESCALA_PIXEL - 1);
      }
    }
  }
}
